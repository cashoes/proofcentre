{{/* Retrieves team members data. If params.google_sheets.enabled is true and sheet_id is set: Fetches the CSV from Google Sheets and parses it into a slice of dicts. Otherwise or on failure: Returns hugo.Data.team (local YAML fallback). */}} {{ $data := slice }} {{ $fallback := hugo.Data.team }} {{ $enabled := site.Params.google_sheets.enabled | default false }} {{ $sheetId := site.Params.google_sheets.sheet_id | default "" }} {{ $tab := site.Params.google_sheets.team_tab | default "Team" }} {{ if and $enabled (ne $sheetId "") }} {{ $url := printf "https://docs.google.com/spreadsheets/d/%s/gviz/tq?tqx=out:csv&sheet=%s" $sheetId (querify "s" $tab | strings.TrimPrefix "s=") }} {{ with try (resources.GetRemote $url) }} {{ with .Err }} {{ warnf "Google Sheets: failed to fetch team from %s: %s. Using local fallback." $url . }} {{ $data = $fallback }} {{ else with .Value }} {{ $rows := .Content | transform.Unmarshal (dict "delimiter" ",") }} {{ if gt (len $rows) 1 }} {{ $headers := index $rows 0 }} {{ range $i, $row := $rows }} {{ if gt $i 0 }} {{ $raw := dict }} {{ range $j, $val := $row }} {{ if lt $j (len $headers) }} {{ $key := index $headers $j | strings.TrimSpace }} {{ $raw = merge $raw (dict $key (strings.TrimSpace $val)) }} {{ end }} {{ end }} {{ $item := dict "name" (index $raw "name") "role" (index $raw "role") "mini_bio" (index $raw "mini_bio") "bio" (index $raw "bio") "orcid" (index $raw "orcid") "image" (index $raw "image") "img_position" (index $raw "img_position" | default "50% 50%") }} {{ $data = $data | append $item }} {{ end }} {{ end }} {{ else }} {{ $data = $fallback }} {{ end }} {{ else }} {{ warnf "Google Sheets: could not retrieve resource for team (%s). Using local fallback." $url }} {{ $data = $fallback }} {{ end }} {{ else }} {{ $data = $fallback }} {{ end }} {{ else }} {{ $data = $fallback }} {{ end }} {{ return $data }}
